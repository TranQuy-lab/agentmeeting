  creds                          -> 1 dong
  password                       -> 0 dong
  api[_-]?key                    -> 0 dong
  BEGIN [A-Z ]*PRIVATE KEY       -> 0 dong
  bearer                         -> 0 dong
  secret                         -> 0 dong
